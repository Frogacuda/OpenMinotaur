# Caverns of Minos source reconstruction (namespace minos). Built standalone for checks
# or added by cores/minos. No dependency on the original binary at build or run time.
cmake_minimum_required(VERSION 3.20)
project(MinosReconstruction LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(MINOS_DEFAULT_CHECKS ON)
else()
  set(MINOS_DEFAULT_CHECKS ${BUILD_TESTING})
endif()
option(MINOS_BUILD_CHECKS "Build standalone Minos reconstruction checks" ${MINOS_DEFAULT_CHECKS})
option(MINOS_CHECK_CROSS_BACKEND_REFERENCES "Also run Windows-versus-glibc bit comparisons as explicit diagnostics" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET neon_source_numeric)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../engine/foundation ${CMAKE_CURRENT_BINARY_DIR}/foundation)
endif()
add_library(mino_minos_rules STATIC cavern_data.cpp cavern_map.cpp cavern_draw.cpp cavern_launch.cpp entity.cpp ship.cpp ship_variants.cpp enemies.cpp game_states.cpp bge.cpp menu_states.cpp)
set_target_properties(mino_minos_rules PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mino_minos_rules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
# Header-only engine pieces (neon/source_random.h: the libc rand stream).
target_include_directories(mino_minos_rules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../../engine/include)
# ship.cpp uses neon::vec2d / neon::colour (engine math, foundation rotation rule): the engine
# library in the main build, its two sources when built standalone.
if(TARGET neon_engine)
  target_link_libraries(mino_minos_rules PUBLIC neon_engine)
else()
  target_sources(mino_minos_rules PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../engine/src/math.cpp
                 ${CMAKE_CURRENT_SOURCE_DIR}/../../engine/src/foundation_rules.cpp)
endif()
# Original float/double operation boundaries: no contraction or reassociation.
target_link_libraries(mino_minos_rules PRIVATE neon_source_numeric)
if(MSVC)
  target_compile_options(mino_minos_rules PRIVATE /W4)
else()
  target_compile_options(mino_minos_rules PRIVATE -Wall -Wextra)
endif()
if(MINOS_BUILD_CHECKS)
  enable_testing()
  add_executable(minos_records_checks records_test.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../../shared/highscores.cpp)
  target_include_directories(minos_records_checks PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../shared)
  target_link_libraries(minos_records_checks PRIVATE mino_minos_rules)
  add_test(NAME minos_local_records COMMAND minos_records_checks ${CMAKE_CURRENT_BINARY_DIR}/records-test-data)
  # Both sets execute original ARM code; only imported math services differ.
  # Windows/MSVC fixtures use UCRT math services; other fixtures use glibc.
  # Preserve supplied glibc fixtures and expose cross-backend differences separately.
  set(MINOS_REFERENCE_BACKEND glibc)
  set(MINOS_SHIP_REFERENCE ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/ship-ios.txt)
  set(MINOS_PLAY_REFERENCE ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/play-ios.txt)
  if(WIN32 AND MSVC)
    set(MINOS_REFERENCE_BACKEND ucrt)
    set(MINOS_SHIP_REFERENCE ${CMAKE_CURRENT_SOURCE_DIR}/diagnostics/ucrt/ship-original-ucrt.txt)
    set(MINOS_PLAY_REFERENCE ${CMAKE_CURRENT_SOURCE_DIR}/diagnostics/ucrt/play-original-ucrt.txt)
  endif()
  add_executable(minos_input_checks input_test.cpp)
  target_link_libraries(minos_input_checks PRIVATE mino_minos_rules)
  add_test(NAME minos_input_source_rules COMMAND minos_input_checks)
  add_executable(minos_ship_variant_checks ship_variants_test.cpp)
  target_link_libraries(minos_ship_variant_checks PRIVATE mino_minos_rules)
  add_test(NAME minos_ship_variants_source_rules COMMAND minos_ship_variant_checks)
  add_executable(minos_cavern_map_checks cavern_map_test.cpp)
  target_link_libraries(minos_cavern_map_checks PRIVATE mino_minos_rules)
  add_test(NAME minos_cavern_lines_original COMMAND minos_cavern_map_checks ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/cavern-lines-ios.txt)
  add_test(NAME minos_cavern_stress_original COMMAND minos_cavern_map_checks --stress ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/cavern-stress-ios.txt)
  add_executable(minos_cavern_draw_checks cavern_draw_test.cpp)
  target_link_libraries(minos_cavern_draw_checks PRIVATE mino_minos_rules)
  add_test(NAME minos_cavern_draw_original COMMAND minos_cavern_draw_checks ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/cavern-draw-ios.txt)
  add_executable(minos_cavern_launch_checks cavern_launch_test.cpp)
  target_link_libraries(minos_cavern_launch_checks PRIVATE mino_minos_rules)
  add_test(NAME minos_cavern_launch_original COMMAND minos_cavern_launch_checks ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/cavern-launch-ios.txt)
  add_executable(minos_entity_checks entity_test.cpp)
  target_link_libraries(minos_entity_checks PRIVATE mino_minos_rules)
  add_test(NAME minos_centity_original COMMAND minos_entity_checks ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/centity-ios.txt)
  add_executable(minos_ship_checks ship_test.cpp)
  target_link_libraries(minos_ship_checks PRIVATE mino_minos_rules)
  add_test(NAME minos_ship_original COMMAND minos_ship_checks ${MINOS_SHIP_REFERENCE})
  set_tests_properties(minos_ship_original PROPERTIES LABELS "translation;${MINOS_REFERENCE_BACKEND}")
  add_executable(minos_play_checks play_test.cpp)
  target_link_libraries(minos_play_checks PRIVATE mino_minos_rules)
  if(EXISTS ${MINOS_PLAY_REFERENCE})
    add_test(NAME minos_play_original COMMAND minos_play_checks ${MINOS_PLAY_REFERENCE})
    set_tests_properties(minos_play_original PROPERTIES LABELS "translation;${MINOS_REFERENCE_BACKEND}")
  endif()
  if(MINOS_CHECK_CROSS_BACKEND_REFERENCES)
    if(NOT (WIN32 AND MSVC))
      message(FATAL_ERROR "These cross-backend diagnostics compare Windows/MSVC output with the supplied glibc fixtures")
    endif()
    add_test(NAME minos_ship_glibc_diagnostic COMMAND minos_ship_checks ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/ship-ios.txt)
    add_test(NAME minos_play_glibc_diagnostic COMMAND minos_play_checks ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/play-ios.txt)
    set_tests_properties(minos_ship_glibc_diagnostic minos_play_glibc_diagnostic PROPERTIES LABELS "cross_backend;diagnostic")
  endif()
endif()
