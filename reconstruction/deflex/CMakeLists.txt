# Deflex source reconstruction (namespace deflex). Built standalone for checks or added by
# cores/deflex. No dependency on the original binary at build or run time.
cmake_minimum_required(VERSION 3.20)
project(DeflexReconstruction LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(DEFLEX_DEFAULT_CHECKS ON)
else()
  set(DEFLEX_DEFAULT_CHECKS ${BUILD_TESTING})
endif()
option(DEFLEX_BUILD_CHECKS "Build standalone Deflex reconstruction checks" ${DEFLEX_DEFAULT_CHECKS})
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET neon_source_numeric)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../engine/foundation ${CMAKE_CURRENT_BINARY_DIR}/foundation)
endif()
add_library(mino_deflex_rules STATIC level_data.cpp cell_grid_addressing.cpp cell_grid_effects.cpp level_setup.cpp deflector_controls.cpp ball_flight.cpp ball_mesh.cpp main_ball.cpp actors.cpp particles.cpp grid_view.cpp world.cpp console.cpp history.cpp frontend.cpp)
set_target_properties(mino_deflex_rules PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mino_deflex_rules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(mino_deflex_rules PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../../engine/include)
if(TARGET neon_engine)
  target_link_libraries(mino_deflex_rules PUBLIC neon_engine)
else()
  target_sources(mino_deflex_rules PRIVATE ../../engine/src/math.cpp ../../engine/src/foundation_rules.cpp)
endif()
target_link_libraries(mino_deflex_rules PRIVATE neon_source_numeric)
if(MSVC)
  target_compile_options(mino_deflex_rules PRIVATE /W4)
else()
  target_compile_options(mino_deflex_rules PRIVATE -Wall -Wextra)
endif()
if(DEFLEX_BUILD_CHECKS)
  enable_testing()
  add_executable(deflex_level_data_checks level_data_test.cpp)
  target_link_libraries(deflex_level_data_checks PRIVATE mino_deflex_rules)
  add_test(NAME deflex_level_records_original COMMAND deflex_level_data_checks ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/level-records-ios.txt)
  add_executable(deflex_cell_addressing_checks cell_grid_addressing_test.cpp)
  target_link_libraries(deflex_cell_addressing_checks PRIVATE mino_deflex_rules)
  add_test(NAME deflex_cell_addressing_original COMMAND deflex_cell_addressing_checks ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/cell-addressing-ios.txt)
  add_executable(deflex_cell_effects_checks cell_grid_effects_test.cpp)
  target_link_libraries(deflex_cell_effects_checks PRIVATE mino_deflex_rules)
  add_test(NAME deflex_cell_effects_source_rules COMMAND deflex_cell_effects_checks)
  add_executable(deflex_level_setup_checks level_setup_test.cpp)
  target_link_libraries(deflex_level_setup_checks PRIVATE mino_deflex_rules)
  add_test(NAME deflex_level_setup_source_rules COMMAND deflex_level_setup_checks)
  add_executable(deflex_ball_flight_checks ball_flight_test.cpp)
  target_link_libraries(deflex_ball_flight_checks PRIVATE mino_deflex_rules)
  add_test(NAME deflex_ball_flight_source_rules COMMAND deflex_ball_flight_checks)
  add_executable(deflex_actor_checks actors_test.cpp)
  target_link_libraries(deflex_actor_checks PRIVATE mino_deflex_rules)
  add_test(NAME deflex_actor_source_rules COMMAND deflex_actor_checks)
  add_executable(deflex_world_checks world_test.cpp)
  target_link_libraries(deflex_world_checks PRIVATE mino_deflex_rules)
  add_test(NAME deflex_live_world COMMAND deflex_world_checks ${CMAKE_CURRENT_SOURCE_DIR}/../../assets/games/deflex/NeonResources.xml)
  add_executable(deflex_frontend_checks frontend_test.cpp)
  target_link_libraries(deflex_frontend_checks PRIVATE mino_deflex_rules)
  add_test(NAME deflex_frontend_history COMMAND deflex_frontend_checks ${CMAKE_CURRENT_SOURCE_DIR}/../../assets/games/deflex/NeonResources.xml)
endif()
