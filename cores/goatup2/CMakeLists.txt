)
    target_compile_definitions(mino_goatup2_backend PRIVATE NEON_GAME=goatup2)
    target_include_directories(mino_goatup2_backend PUBLIC ${GU2_BINDING} ${GU2_RESOURCE_BINDING}
        ${PROJECT_SOURCE_DIR}/include)
    target_link_libraries(mino_goatup2_backend PUBLIC mino_goatup2_rules neon_renderer PRIVATE mino_baked_assets)
    add_library(mino_core_goatup2 STATIC EXCLUDE_FROM_ALL goatup2_core.cpp)
    target_compile_definitions(mino_core_goatup2 PRIVATE NEON_GAME=goatup2)
    target_link_libraries(mino_core_goatup2 PUBLIC mino_goatup2_backend)

# Standalone CTest registrations must be runnable after an ordinary build.
# Collection builds retain their explicit check-build dependency policy.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    get_property(goatup2_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS goatup2_targets)
        if(target MATCHES "^mino_goatup2_.*_checks$|^mino_goatup2_runtime$")
            set_target_properties(${target} PROPERTIES EXCLUDE_FROM_ALL FALSE)
        endif()
    endforeach()
endif()
