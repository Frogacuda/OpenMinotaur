if(NOT ANDROID)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    file(GLOB_RECURSE GAME_ASSET_FILES CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/assets/games/*")
    set(ASSET_ARCHIVE "${CMAKE_CURRENT_BINARY_DIR}/game-assets.zip")
    if(WIN32)
        enable_language(RC)
        set(ASSET_FORMAT windows)
        set(ASSET_EMBED "${CMAKE_CURRENT_BINARY_DIR}/game-assets.rc")
    else()
        set(ASSET_EMBED "${CMAKE_CURRENT_BINARY_DIR}/game-assets.cpp")
        if(APPLE)
            set(ASSET_FORMAT apple)
        else()
            set(ASSET_FORMAT elf)
        endif()
    endif()
    add_custom_command(OUTPUT "${ASSET_EMBED}" "${ASSET_ARCHIVE}"
        COMMAND Python3::Interpreter "${PROJECT_SOURCE_DIR}/tools/bundle_game_assets.py"
            --source "${PROJECT_SOURCE_DIR}/assets/games" --archive "${ASSET_ARCHIVE}"
            --embed "${ASSET_EMBED}" --format "${ASSET_FORMAT}"
        DEPENDS ${GAME_ASSET_FILES} assets/GAME-ASSETS.json tools/bundle_game_assets.py
        VERBATIM)
    target_sources(${APP_TARGET} PRIVATE "${ASSET_EMBED}")
    target_sources(mino_launcher_checks PRIVATE "${ASSET_EMBED}")
    if(TARGET mino_app_resume_checks)
        target_sources(mino_app_resume_checks PRIVATE "${ASSET_EMBED}")
    endif()
    add_executable(mino_bundled_asset_checks EXCLUDE_FROM_ALL tests/bundled_assets.cpp "${ASSET_EMBED}")
    target_link_libraries(mino_bundled_asset_checks PRIVATE mino_game_assets)
endif()
