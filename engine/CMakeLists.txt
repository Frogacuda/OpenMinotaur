# This directory can also be configured independently, without SDL or a GPU.
cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NeonEngine LANGUAGES C CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    add_library(mino_pugixml STATIC ../third_party/pugixml/pugixml.cpp)
    target_include_directories(mino_pugixml PUBLIC ../third_party/pugixml)
    set(NEON_STANDALONE ON)
endif()

# Namespace bindings carry no global runtime state through this interface target.
add_library(neon_module_bindings INTERFACE)
target_include_directories(neon_module_bindings INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}/bindings
    ${CMAKE_CURRENT_SOURCE_DIR}/bindings/common)

add_subdirectory(foundation)
add_library(neon_image STATIC src/image_decode.cpp)
target_include_directories(neon_image PRIVATE ../third_party/stb)
set_target_properties(neon_image PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
add_library(neon_engine STATIC src/resource_catalog.cpp src/math.cpp src/effect_dna.cpp src/render_state.cpp src/sample_audio.cpp)
target_include_directories(neon_engine PUBLIC include)
target_link_libraries(neon_engine PUBLIC neon_foundation Threads::Threads PRIVATE mino_pugixml)
target_link_libraries(neon_engine PRIVATE neon_source_numeric)

if(NOT TARGET mino_glad)
    add_library(mino_glad STATIC ../third_party/glad/src/gl.c)
    target_include_directories(mino_glad PUBLIC ../third_party/glad/include)
endif()
add_library(neon_renderer STATIC src/render_device.cpp src/graphics_lifecycle.cpp)
target_include_directories(neon_renderer PUBLIC include PRIVATE ../third_party/stb)
target_link_libraries(neon_renderer PUBLIC neon_engine mino_glad neon_image neon_module_bindings)

option(NEON_BUILD_CHECKS "Build shared engine checks" ${BUILD_TESTING})
if(NEON_STANDALONE OR NEON_BUILD_CHECKS)
    enable_testing()
    add_executable(runtime_mechanics_checks tests/runtime_mechanics.cpp)
    target_link_libraries(runtime_mechanics_checks PRIVATE neon_engine)
    add_test(NAME controller-touch-and-list-lifecycle COMMAND runtime_mechanics_checks)
    add_executable(sample_audio_checks tests/sample_audio.cpp)
    target_link_libraries(sample_audio_checks PRIVATE neon_engine)
    add_test(NAME original-sample-audio COMMAND sample_audio_checks --fixture ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/audio-arithmetic.tsv)
    add_executable(effect_runtime_checks tests/effect_runtime.cpp)
    target_link_libraries(effect_runtime_checks PRIVATE neon_engine)
    add_test(NAME effect-execution-contract COMMAND effect_runtime_checks)
    add_executable(input_contract_checks tests/input.cpp)
    target_include_directories(input_contract_checks PRIVATE include)
    add_test(NAME native-input-and-menu-services COMMAND input_contract_checks)
    add_executable(sprite_draw_checks tests/sprite_draw.cpp)
    target_link_libraries(sprite_draw_checks PRIVATE neon_engine)
    add_test(NAME sprite-emission-and-batching COMMAND sprite_draw_checks ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/sprite-quads.tsv)
    add_executable(entity_contact_checks tests/entity_contact.cpp)
    target_link_libraries(entity_contact_checks PRIVATE neon_engine)
    add_test(NAME early-contact-callback COMMAND entity_contact_checks)
    add_executable(foundation_rules_checks tests/foundation_rules.cpp)
    target_link_libraries(foundation_rules_checks PRIVATE neon_foundation)
    add_test(NAME original-foundation-variants COMMAND foundation_rules_checks ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/palette-pulses.tsv)
    add_executable(render_device_parity tests/render_device_parity.cpp tests/legacy_render_device.cpp)
    target_include_directories(render_device_parity PRIVATE ../third_party/stb)
    target_link_libraries(render_device_parity PRIVATE neon_renderer)
    add_test(NAME render-device COMMAND render_device_parity)
    add_executable(resource_parity tests/resource_parity.cpp tests/legacy_resources.cpp)
    target_compile_definitions(resource_parity PRIVATE NEON_GAME=reference)
    target_link_libraries(resource_parity PRIVATE neon_engine mino_pugixml)
    file(GLOB MANIFEST_FIXTURES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/*.xml")
    foreach(manifest IN LISTS MANIFEST_FIXTURES)
        get_filename_component(game "${manifest}" NAME_WE)
        add_test(NAME "resources-${game}" COMMAND resource_parity "${manifest}")
    endforeach()
    add_executable(math_parity tests/math_parity.cpp tests/legacy_base.cpp)
    target_compile_definitions(math_parity PRIVATE NEON_GAME=reference)
    target_link_libraries(math_parity PRIVATE neon_engine)
    add_test(NAME foundation-math COMMAND math_parity)
    add_executable(render_state_parity tests/render_state_parity.cpp tests/legacy_shader_cycle.cpp)
    target_link_libraries(render_state_parity PRIVATE neon_engine)
    add_test(NAME render-state COMMAND render_state_parity)
    add_executable(effect_dna_parity tests/effect_dna_parity.cpp tests/legacy_effect_dna.cpp)
    target_link_libraries(effect_dna_parity PRIVATE neon_engine)
    file(GLOB EFFECT_FIXTURES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/*.ox")
    foreach(grid IN LISTS EFFECT_FIXTURES)
        get_filename_component(game "${grid}" NAME_WE)
        add_test(NAME "effect-graphs-${game}" COMMAND effect_runtime_checks "${grid}")
        add_test(NAME "effects-${game}" COMMAND effect_dna_parity "${grid}")
    endforeach()
endif()

if(NEON_STANDALONE OR NEON_BUILD_CHECKS)
    get_property(engine_test_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(check IN LISTS engine_test_targets)
        get_target_property(check_type ${check} TYPE)
        if(check_type STREQUAL "EXECUTABLE")
            target_link_libraries(${check} PRIVATE neon_source_numeric)
        endif()
    endforeach()
endif()
