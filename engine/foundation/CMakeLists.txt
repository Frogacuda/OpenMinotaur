cmake_minimum_required(VERSION 3.20)
if(NOT TARGET neon_source_numeric)
  # Shared source algorithms require the original operation/rounding boundaries,
  # including consumers that instantiate the engine's header implementations.
  add_library(neon_source_numeric INTERFACE)
  if(MSVC)
    target_compile_options(neon_source_numeric INTERFACE /fp:strict)
  else()
    target_compile_options(neon_source_numeric INTERFACE -fno-fast-math -ffp-contract=off)
  endif()
endif()
if(NOT TARGET neon_foundation)
  add_library(neon_foundation STATIC ../src/foundation_rules.cpp)
  target_include_directories(neon_foundation PUBLIC ../include)
  target_compile_features(neon_foundation PUBLIC cxx_std_17)
  set_target_properties(neon_foundation PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(neon_foundation PUBLIC neon_source_numeric)
  if(MSVC)
    target_compile_options(neon_foundation PRIVATE /W4)
  else()
    target_compile_options(neon_foundation PRIVATE -Wall -Wextra)
  endif()
endif()
